// Myanmar Watch Proxy Worker — avec cache KV 10min // KV namespace à créer: MYANMAR_CACHE // Binding: variable name = MYANMAR_CACHE const CACHE_TTL = 600; // 10 minutes en secondes export default { async fetch(request, env) { const url = new URL(request.url); // CORS preflight if (request.method === 'OPTIONS') { return new Response(null, { headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, OPTIONS', 'Access-Control-Allow-Headers': '*', 'Access-Control-Max-Age': '86400', } }); } const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, OPTIONS', 'Access-Control-Allow-Headers': '*', }; // Proxy RSS const targetUrl = url.searchParams.get('url'); if (!targetUrl) { return new Response('Missing ?url= parameter', { status: 400, headers: corsHeaders }); } // Clé de cache = URL encodée (max 512 chars pour KV) const cacheKey = 'rss:' + targetUrl.slice(0, 500); // 1. Vérifier le cache KV if (env.MYANMAR_CACHE) { try { const cached = await env.MYANMAR_CACHE.get(cacheKey); if (cached) { return new Response(cached, { headers: { ...corsHeaders, 'Content-Type': 'application/xml; charset=utf-8', 'X-Cache': 'HIT', } }); } } catch (e) { // KV error — continue without cache } } // 2. Fetch la vraie source try { const resp = await fetch(targetUrl, { headers: { 'User-Agent': 'Mozilla/5.0 (compatible; MyanmarWatch/1.0)', 'Accept': 'application/rss+xml, application/xml, text/xml, */*', }, cf: { cacheTtl: CACHE_TTL, cacheEverything: true }, // Cache Cloudflare edge aussi }); if (!resp.ok) { return new Response(`Upstream error: ${resp.status}`, { status: resp.status, headers: corsHeaders }); } const body = await resp.text(); // 3. Stocker dans KV avec TTL if (env.MYANMAR_CACHE && body.length > 0) { try { await env.MYANMAR_CACHE.put(cacheKey, body, { expirationTtl: CACHE_TTL }); } catch (e) { // KV write error — ignore } } return new Response(body, { headers: { ...corsHeaders, 'Content-Type': resp.headers.get('Content-Type') || 'application/xml; charset=utf-8', 'X-Cache': 'MISS', } }); } catch (err) { return new Response(`Fetch error: ${err.message}`, { status: 502, headers: corsHeaders }); } } };
